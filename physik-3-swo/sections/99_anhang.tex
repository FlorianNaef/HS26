\subsection{Differentialgleichungen}\label{Differentialgleichungen}

Übersicht aller in dieser Zusammenfassung auftretenden Differentialgleichungen (DGL) inklusive Lösungsansätzen.

\subsubsection{Ungedämpfte Schwingungen -- homogene DGL}

$$ \boxed{ \text{DGL: } \ddot{x} = -\omega^2 \cdot x \qquad \text{Lösung: } x(t) = A \cdot \sin(\omega \, t + \varphi) } $$

\renewcommand{\arraystretch}{1.5}
\begin{ctabular}{l M}
    \textbf{System}               & \bm{\omega^2} \\
    \hline
    Federpendel                   & \omega^2 = \dfrac{k}{m} \\
    Fadenpendel                   & \omega^2 = \dfrac{g}{l} \\
    Dreh- / Torsionspendel        & \omega^2 = \dfrac{c^*}{J} \\
    Physikalisches Pendel         & \omega^2 = \dfrac{g}{L^*} = \dfrac{g \cdot a \cdot m}{J_A} \\
\end{ctabular}
\renewcommand{\arraystretch}{1}

\subsubsection{Gedämpfte Schwingungen -- homogene DGL}

\[ \boxed{ \text{DGL: } \quad \ddot{x}(t) + 2 \delta \, \dot{x}(t) + \omega^2 \, x(t) = 0 } \qquad \boxed{ D = \frac{\delta}{\omega} = \frac{\kappa}{2 \omega} } \]

\para{Lösungsweg} Exponentialansatz $x(t) = e^{\lambda t}$ \textrightarrow\ charakteristische Gleichung:

$$ \lambda^2 + 2 \, \delta \, \lambda + \omega^2 = 0 \qquad \Rightarrow \qquad \lambda_{1,2} = -\delta \pm \sqrt{\delta^2 - \omega^2} $$

\resizebox{\columnwidth}{!}{
    \begin{tabular}{lll}
        \toprule
        $D > 1$ & $x(t) = A e^{\lambda_1 t} + B e^{\lambda_2 t}$, \ $\lambda_{1,2} = -\omega \left( D \pm \sqrt{D^2 - 1} \right)$ & Aperiodisch \\
        \midrule
        $D < 1$ & $x(t) = A e^{-\delta t} \cdot \cos \left( \Omega t + \varphi \right)$, \ $\Omega^2 = \omega^2 \left( 1 - D^2 \right)$ & Periodisch gedämpft \\
        \midrule
        $D = 1$ & $x(t) = (A + Bt) \cdot e^{-\delta t}$ & Grenzfall \\
        \bottomrule
    \end{tabular}
}

\subsubsection{Fremderregte Schwingungen -- inhomogene DGL}

\[ \boxed{ \text{DGL: } \quad m \, \ddot{y} + b \, \dot{y} + c \, y = F(t) } \]

\para{Struktur der Lösung} $y(t) = \underbrace{A(\omega) \cdot \sin(\omega t - \varphi)}_{y_p(t)} + \underbrace{B \cdot e^{-\delta t} \sin(\Omega_d t + \varphi_0)}_{y_h(t)}$ \\
\textrightarrow\ Partikuläre Lösung $y_p$ (erzwungene Schwingung) + homogene Lösung $y_h$ (abklingende Eigenschwingung).

\renewcommand{\arraystretch}{1.7}
\begin{ctabular}{l M}
    \textbf{Erregung}        & \bm{F(t)} \\
    \hline
    Krafterregung             & F_0 \cdot \sin(\omega t) \\
    Dämpfererregung           & b \, \omega \, u_0 \cdot \cos(\omega t) \\
    Stützenerregung           & -\,m \, \omega^2 \, u_0 \cdot \sin(\omega t) \\
    Unwuchterregung           & -\,m_R \, \omega^2 \, e \cdot \sin(\omega t) \\
    Schwingkreis (RLC)        & \omega \, U_0 \cdot \sin \left( \omega t + \frac{\pi}{2} \right) \\
\end{ctabular}
\renewcommand{\arraystretch}{1}

$$ \boxed{ \text{Resonanz:} \quad \omega_r = \omega_0 \sqrt{1 - 2 D^2} } \qquad \boxed{ \text{Güte:} \quad Q = \frac{1}{2D} = \frac{\omega_0}{\Delta \omega} } $$

\subsubsection{Gekoppelte Pendel -- DGL-System}

$$ J_1 \, \ddot{\varphi}_1 = -m_1 \, g \, a_1 \, \varphi_1 + c \, h^2 \, (\varphi_2 - \varphi_1) \qquad \qquad J_2 \, \ddot{\varphi}_2 = -m_2 \, g \, a_2 \, \varphi_2 - c \, h^2 \, (\varphi_2 - \varphi_1) $$

\para{Lösungsweg} Übergang auf Normalkoordinaten: $\Phi_+ = \varphi_1 + \varphi_2$ (symmetrisch, $\ddot{\Phi}_+ = -\omega^2 \, \Phi_+$) und $\Phi_- = \varphi_2 - \varphi_1$ (antisymmetrisch, $\ddot{\Phi}_- = -(\omega^2 + 2k) \, \Phi_-$). Allgemeine Lösung: Linearkombination beider Eigenschwingungen \textrightarrow\ Schwebung.

\subsubsection{Wellengleichung -- partielle DGL}

\[ \boxed{ \text{1D: } \frac{\partial^2 \xi}{\partial x^2} = \frac{1}{u^2} \frac{\partial^2 \xi}{\partial t^2} } \qquad \boxed{ \text{3D: } \frac{\partial^2 \xi}{\partial x^2} + \frac{\partial^2 \xi}{\partial y^2} + \frac{\partial^2 \xi}{\partial z^2} = \frac{1}{u^2} \frac{\partial^2 \xi}{\partial t^2} } \]

\renewcommand{\arraystretch}{1.5}
\begin{tabular}{ll}
    Sinusansatz (1D)      & $\xi(x, t) = \xi_0 \cdot \sin(\omega t - k x)$ \\

    Kugelwelle            & $\xi(t, r) = \dfrac{1}{r} \, \xi_0 \, e^{\jimg \, (\omega t - k r)}$ \\
\end{tabular}
